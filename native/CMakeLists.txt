cmake_minimum_required(VERSION 3.20)
project(ESPmonitorNative LANGUAGES C CXX)
enable_testing()
add_executable(quotamon_protocol_test ../tests/frame_test.cpp)
add_test(NAME ble-framing COMMAND quotamon_protocol_test)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
if(MSVC AND NOT EXISTS "$ENV{VCToolsInstallDir}/bin/Hostx64/x64/1033/clui.dll")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "참고: 포함 파일: ")
endif()
set(LVGL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/lvgl")
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_ROOT}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC "${LVGL_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE=1 LV_LVGL_H_INCLUDE_SIMPLE=1)
file(GLOB GENERATED_SOURCES CONFIGURE_DEPENDS "generated/*.c")
add_library(espmonitor_ui STATIC ui.cpp sprite_decoder.cpp ${GENERATED_SOURCES})
target_link_libraries(espmonitor_ui PUBLIC lvgl)
target_include_directories(espmonitor_ui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(quotamon_runtime_metrics_test ../tests/runtime_metrics_test.cpp ../firmware/main/runtime_metrics.cpp)
target_include_directories(quotamon_runtime_metrics_test PRIVATE ../tests/runtime_stubs ../protocol ../third_party/nlohmann)
target_link_libraries(quotamon_runtime_metrics_test PRIVATE lvgl)
add_test(NAME runtime-metrics COMMAND quotamon_runtime_metrics_test)
if(MSVC)
  target_compile_options(quotamon_runtime_metrics_test PRIVATE /utf-8 /W4)
  target_compile_options(lvgl PRIVATE /utf-8)
  target_compile_options(espmonitor_ui PRIVATE /utf-8 /W4)
endif()
if(WIN32)
  add_executable(espmonitor_sim simulator_win32.cpp)
  target_link_libraries(espmonitor_sim PRIVATE espmonitor_ui user32 gdi32)
  target_compile_options(espmonitor_sim PRIVATE /utf-8 /W4)
endif()
