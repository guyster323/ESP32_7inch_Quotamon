cmake_minimum_required(VERSION 3.16)
set(PROJECT_VER "0.1.0")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(quotamon_ble)
idf_build_set_property(COMPILE_DEFINITIONS "LV_CONF_PATH=${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" APPEND)

# Keep forced headers attached to their option when PlatformIO reorders CMake flags.
foreach(component espressif__esp_lvgl_adapter lvgl__lvgl)
  idf_component_get_property(component_library ${component} COMPONENT_LIB)
  foreach(property COMPILE_OPTIONS INTERFACE_COMPILE_OPTIONS)
    get_target_property(options ${component_library} ${property})
    if(options)
      set(joined_options)
      set(forced_header FALSE)
      foreach(option IN LISTS options)
        if(forced_header)
          list(APPEND joined_options "-Wp,-include,${option}")
          set(forced_header FALSE)
        elseif(option STREQUAL "-include")
          set(forced_header TRUE)
        else()
          list(APPEND joined_options "${option}")
        endif()
      endforeach()
      set_property(TARGET ${component_library} PROPERTY ${property} "${joined_options}")
    endif()
  endforeach()
endforeach()
